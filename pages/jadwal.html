Integrasi Google Spreadsheet

URL Google Apps Script berikut sudah tersimpan otomatis. Anda dapat memperbaruinya kapan saja jika membuat Web App baru.

Pastikan Kode Apps Script Anda Seperti Ini: 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var data = [];
  for (var i = 1; i < rows.length; i++) {
    if(rows[i][1]) {
      var d = rows[i][2];
      if (d instanceof Date) {
        d = Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd");
      }
      data.push({ no: rows[i][0], nama: rows[i][1], tanggal: d });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: data }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var params = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var lastRow = sheet.getLastRow();
  var nextNo = lastRow;
  sheet.appendRow([nextNo, params.nama, params.tanggal]);
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}
